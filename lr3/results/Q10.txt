Q10. Клієнти з бронюваннями

SELECT u.user_id, u.first_name, u.last_name
FROM `user` u
WHERE EXISTS (
    SELECT 1 FROM reservation r WHERE r.user_id = u.user_id
)
ORDER BY u.user_id;
