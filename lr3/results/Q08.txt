Q08. Кількість позицій за кухарями

SELECT c.chef_id, c.name, COUNT(oi.order_item_id) AS item_count
FROM chef c
LEFT JOIN order_item oi ON oi.chef_id = c.chef_id
GROUP BY c.chef_id, c.name
ORDER BY item_count DESC, c.chef_id;
