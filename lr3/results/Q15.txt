Q15. Клієнти з двома або більше замовленнями

SELECT u.user_id, u.first_name, u.last_name
FROM `user` u
WHERE (SELECT COUNT(*) FROM `order` o WHERE o.user_id = u.user_id) >= 2
ORDER BY u.user_id;
