Q06. Замовлення та їхні клієнти

SELECT o.order_id, CONCAT(u.first_name, ' ', u.last_name) AS customer,
       o.order_status
FROM `order` o
JOIN `user` u ON u.user_id = o.user_id
ORDER BY o.order_id
LIMIT 10;
