Q05. Три найдорожчі напої

SELECT drink_id, name, price
FROM drink
ORDER BY price DESC, drink_id
LIMIT 3;

| drink_id | name           | price  |

|        4 | Лате           |  65.00 |
|        7 | Какао          |  60.00 |
