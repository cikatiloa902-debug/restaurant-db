Q16. Замовлення без оплати

SELECT o.order_id, o.order_status
FROM `order` o
WHERE NOT EXISTS (
    SELECT 1 FROM `transaction` t WHERE t.order_id = o.order_id
)
ORDER BY o.order_id;
