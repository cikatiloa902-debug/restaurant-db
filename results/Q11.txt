Q11. Клієнти без замовлень

SELECT u.user_id, u.first_name, u.last_name
FROM `user` u
WHERE NOT EXISTS (
    SELECT 1 FROM `order` o WHERE o.user_id = u.user_id
)
ORDER BY u.user_id;
